private static async Task<string> DescribePriceAsync(string sku)
{
    decimal? price = await FindPriceOrNullAsync(sku);
    if (price != null)
    {
        if (price > 0)
        {
            try
            {
                SavePriceOrThrow(sku, price.Value);
                Console.WriteLine($"Saved {sku} at {price:F2}");
                return $"Success: {sku} costs {price:F2}";
            }
            catch (InvalidOperationException exception)
            {
                return $"Expected Failure: {exception.Message}";
            }
        }
        else
        {
            return $"Expected Failure: The price for '{sku}' must be positive.";
        }
    }

    return $"Expected Failure: No price for '{sku}'.";
}
private static Task<string> DescribePriceAsync(string sku) =>
    FindPriceAsync(sku)
        .ToResultAsync(new Error($"No price for '{sku}'."))
        .EnsureAsync(price => price > 0, $"The price for '{sku}' must be positive.")
        .TapAsync(price => SavePrice(sku, price))
        .TapAsync(price => Console.WriteLine($"Saved {sku} at {price:F2}"))
        .MatchAsync(price => $"Success: {sku} costs {price:F2}",
            error => $"Expected Failure: {error.Message}");

public Task ASavedDocumentOpensOfflineAsAScenario()
{
    Document expected = new("handbook", "Offline-ready field handbook");
    FakeDocumentSource source = new(expected);
    FakeInMemoryDocumentCache cache = new();
    DocumentReader reader = new(new FakeOfflineConnectivity(), source, cache);
    DocumentReadResult? result = null;

    return Scenario.Define("A returning offline reader opens a saved handbook")
        .Given("An offline reader with a saved document", () => cache.Seed(expected))
        .When("The reader opens the handbook",
            async token => result = await reader.OpenAsync("handbook", token))
        .Then("The saved document is returned", () => Assert.Equal(expected, result?.Document))
        .And("The web source is not contacted", () => Assert.Equal(0, source.DownloadCalls))
        .VerifyAsync();
}
